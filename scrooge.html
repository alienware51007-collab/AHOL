C — half-coded zero handler
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX 256

static char op_token = '$';

/* Half-coded: turns every 0 into 0* ; leaves $ untouched */
char *clinches_zeros(const char *src) {
    char *dst = malloc(MAX);
    int j = 0;
    for (int i = 0; src[i] && j < MAX-2; i++) {
        if (src[i] == '0') {
            dst[j++] = '0';
            dst[j++] = '*';
        } else if (src[i] == op_token) {
            dst[j++] = op_token;
        } else {
            dst[j++] = src[i];
        }
    }
    dst[j] = '\\0';
    return dst;
}

void demo(void) {
    const char *top    = "010$10";
    const char *bottom = "0$1001";
    char *t = clinches_zeros(top);
    char *b = clinches_zeros(bottom);
    printf("PCP top:    %s\\n", t);
    printf("PCP bottom: %s\\n", b);
    free(t); free(b);
}
Python — cookies + parent model
# 3 lines of cookies that collect the clinched zeros
import http.cookies
c = http.cookies.SimpleCookie(); c['zeros'] = '0*0*0*'; c['op'] = '$'
print(c.output())   # Set-Cookie headers ready for storage

# Parent model for the $ function — keeps the operational token mutable
class ParentPCP:
    def dollar(self, token: str) -> str:
        return token if token == '$' else token

class LinuxMutation(ParentPCP):
    def dollar(self, token: str) -> str:
        if token == '$':
            return '$→'          # figurative mutation
        return super().dollar(token)

m = LinuxMutation()
print(m.dollar('$'))
